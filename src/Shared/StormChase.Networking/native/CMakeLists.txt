cmake_minimum_required(VERSION 3.15)
project(StormChaseNativeTransport LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIB ON CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ENABLE_ICE OFF CACHE BOOL "" FORCE)
set(Protobuf_USE_STATIC_LIBS ON CACHE BOOL "" FORCE)
if(WIN32)
  set(USE_CRYPTO BCrypt CACHE STRING "" FORCE)
endif()
if(NOT GNS_SOURCE_DIR)
  message(FATAL_ERROR "Set GNS_SOURCE_DIR to the pinned GameNetworkingSockets source checkout.")
endif()
add_subdirectory("${GNS_SOURCE_DIR}" gns)
# Trust roots are installed explicitly by the application's trusted HTTPS discovery.
# This upstream switch removes Valve's hardcoded CA; no network packet may add roots.
target_compile_definitions(GameNetworkingSockets_s PRIVATE STEAMNETWORKINGSOCKETS_ALLOW_DYNAMIC_SELFSIGNED_CERTS)
add_library(stormchase_gns SHARED stormchase_gns.cpp)
target_link_libraries(stormchase_gns PRIVATE GameNetworkingSockets::static)
target_compile_definitions(stormchase_gns PRIVATE VALVE_CRYPTO_ENABLE_25519)
target_include_directories(stormchase_gns PRIVATE "${GNS_SOURCE_DIR}/src/common" "${GNS_SOURCE_DIR}/src/public" "${CMAKE_CURRENT_BINARY_DIR}/gns/src")
if(MINGW)
  target_link_options(stormchase_gns PRIVATE -static-libstdc++ -static-libgcc -static)
endif()
set_target_properties(stormchase_gns PROPERTIES PREFIX "")
